NON-ANSI METHOD
-----------------
INNER JOIN
---------
SELECT C.CID,C.CNAME,S.CID,S.SNAME
FROM COURSE C,STUDENTS S
WHERE C.CID=S.CID;

ANSI METHOD
-------------
SELECT C.CID,C.CNAME,S.CID,S.SNAME
FROM COURSE C INNER JOIN STUDENTS S
ON C.CID=S.CID;


NON-ANSI METHOD
-----------------
LEFT JOIN
---------
SELECT C.CID,C.CNAME,S.CID,S.SNAME
FROM COURSE C,STUDENTS S
WHERE C.CID=S.CID(+);--PREDICATE

ANSI METHOD
-------------
SELECT C.CID,C.CNAME,S.CID,S.SNAME
FROM COURSE C LEFT JOIN STUDENTS S
ON C.CID=S.CID;

NON-ANSI METHOD
-----------------
RIGHT JOIN
---------
SELECT C.CID,C.CNAME,S.CID,S.SNAME
FROM COURSE C,STUDENTS S
WHERE C.CID(+)=S.CID;--PREDICATE

ANSI METHOD
-------------
SELECT C.CID,C.CNAME,S.CID,S.SNAME
FROM COURSE C RIGHT JOIN STUDENTS S
ON C.CID=S.CID;


FULL JOIN
---------

ANSI METHOD
-------------
SELECT C.CID,C.CNAME,S.CID,S.SNAME
FROM COURSE C FULL JOIN STUDENTS S
ON C.CID=S.CID;

CROSS JOIN
---------

ANSI METHOD
-------------
SELECT C.CID,C.CNAME,S.CID,S.SNAME
FROM COURSE C CROSS JOIN STUDENTS S;
