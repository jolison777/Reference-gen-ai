General Functions
---------------------
1 Nvl
------------
it is used to handle null values
it accepts 2 paramters
if the first parameter is not null, then it prints first parameter value
if the first parameter is null, then it prints second parameter value
select commission_pct,nvl(commission_pct,0) from employees;
select commission_pct,nvl(commission_pct,100) from employees;

Coalesce
---------------
it accepts n number of parameters
it prints first not null value

select coalesce(100,null,null,0) from dual;
select coalesce(null,0,100,null,null,0) from dual;

Case
---------
SELECT EMPLOYEE_ID,'AIMORE' FROM EMPLOYEES;--AIMORE IS DERVIED COLUMN

SALARY >= 17000  'HIGH'
SALARY BETWEN 10000 AND 17000  'MEDIUM'
SALARY <= 10000  'LOW'

SELECT EMPLOYEE_ID,FIRST_NAME,SALARY,CASE WHEN SALARY >= 17000 THEN 'HIGH'
                                          WHEN SALARY BETWEEN 10000 AND 17000 THEN 'MEDIUM'
                                          WHEN SALARY <= 10000 THEN 'LOW'
                                     END
FROM EMPLOYEES;

SELECT EMPLOYEE_ID,FIRST_NAME,SALARY,CASE WHEN SALARY >= 17000 THEN 'HIGH'
                                          WHEN SALARY BETWEEN 10000 AND 17000 THEN 'MEDIUM'
                                          ELSE 'LOW'
                                    END
FROM EMPLOYEES;

SELECT EMPLOYEE_ID,FIRST_NAME,SALARY,CASE WHEN SALARY >= 17000 THEN 'HIGH'
                                          WHEN SALARY BETWEEN 10000 AND 17000 THEN 'MEDIUM'
                                          ELSE 'LOW'
                                     END AS SAL_RANGE
FROM EMPLOYEES;

SELECT *
FROM EMPLOYEES
WHERE SALARY BETWEEN 10000 AND 17000 ;
